cmake_minimum_required(VERSION 3.31.6)
project(coder_wan LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_GGML_LIB OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_ROOT ${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/stable-diffusion.cpp)
add_subdirectory(${SD_ROOT} build-sd)
add_library(coder-wan SHARED wan_jni.cpp)
target_include_directories(coder-wan PRIVATE ${SD_ROOT}/include)
target_link_libraries(coder-wan PRIVATE stable-diffusion android log)
# Keep this runtime's GGML symbols private; llama.cpp has a different GGML version.
target_link_options(coder-wan PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic")
